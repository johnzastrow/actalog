# ActaLog - Reverse Proxy Configuration
# This Caddyfile configures two PWA applications:
# 1. ActaLog (PWA) at https://al.fluidgrid.site (internal port 3000)
# 2. Mealie (PWA) at https://recipe.fluidgrid.site (internal port 9925)

# Global options for debugging
{
        debug
        # Enable admin API for debugging
        admin 0.0.0.0:2019
}

# ActaLog Configuration
al.fluidgrid.site {
        # Reverse proxy to ActaLog frontend running on port 3000
        reverse_proxy localhost:8080

        # Enhanced logging with human-readable format and debugging info
        log {
                output file /var/log/caddy/al.access.log {
                        roll_size 1MB # Create new file when size exceeds 1MB
                        roll_keep 5 # Keep at most 5 rolled files
                }
                format console # Human-readable format instead of JSON
                level DEBUG # Enable debug-level logging
        }

        # Response headers for security
        header {
                # Security headers
                Strict-Transport-Security "max-age=31536000; includeSubDomains"
                X-Frame-Options "SAMEORIGIN"
                X-Content-Type-Options "nosniff"
                X-XSS-Protection "1; mode=block"
                Referrer-Policy "strict-origin-when-cross-origin"
        }

        # Compression for better performance (important for PWA)
        encode gzip

        # Timeouts for reverse proxy
        reverse_proxy localhost:8080 {
                # Set timeouts for backend communication
                transport http {
                        response_header_timeout 30s
                        read_timeout 1m
                        write_timeout 1m
                }

                # Health check
                health_uri /health
                health_interval 30s
                health_timeout 5s

                # Forward client information
                header_up Host {host}
                header_up X-Real-IP {remote_host}
                header_up X-Forwarded-For {remote_host}
                header_up X-Forwarded-Proto {scheme}

                # Debug headers to verify proxy is working
                header_down X-Debug-Proxy "ActaLog-Caddy"
                header_down X-Debug-Backend "localhost:4173"
        }

        # Diagnostic endpoint to test Caddy is running
        handle /caddy-test {
                respond "Caddy is serving al.fluidgrid.site - {time.now}" 200
        }
}
